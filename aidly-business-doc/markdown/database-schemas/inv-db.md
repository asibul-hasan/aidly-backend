## Physical Count (INV_2006) — added 2026-08-14

```sql
-- Stocktake sheet. Posting does NOT write stock: it builds an
-- inv_stock_adjustment (adjustment_type 5 = Count) and puts that through the
-- posting engine, so a count reaches the ledger by the same path as every other
-- movement and is reversible the same way.
CREATE TABLE IF NOT EXISTS inv_physical_count (
    count_no      bigint generated by default as identity,
    company_no    bigint      not null,
    branch_no     bigint      not null,
    count_id      varchar(40) not null,
    count_date    date        not null,
    warehouse_no  bigint      not null,
    count_type    smallint    not null default 1,   -- 1=Full 2=Cycle 3=Spot
    status        smallint    not null default 1,   -- 1=Draft 2=Counting 3=Review 4=Posted 5=Cancelled
    freeze_stock  smallint    not null default 0,   -- advisory only; not enforced by the engine
    variance_adjustment_no bigint,                  -- the adjustment posting created
    remarks       text,
    is_active smallint not null default 1, is_deleted smallint not null default 0,
    created_by bigint not null, created_at timestamptz not null default CURRENT_TIMESTAMP,
    updated_by bigint, updated_at timestamptz, deleted_by bigint, deleted_at timestamptz,
    row_version bigint not null default 1,
    primary key (count_no)
);
CREATE UNIQUE INDEX IF NOT EXISTS uq_inv_count_id on inv_physical_count (branch_no, count_id) WHERE is_deleted = 0;
CREATE INDEX IF NOT EXISTS idx_inv_count_warehouse on inv_physical_count (company_no, warehouse_no, count_date) WHERE is_deleted = 0;

CREATE TABLE IF NOT EXISTS inv_physical_count_dtl (
    count_dtl_no bigint generated by default as identity,
    count_no     bigint        not null references inv_physical_count (count_no) on delete cascade,
    line_no      integer       not null,
    product_no   bigint        not null,
    variant_no   bigint,
    batch_no     bigint,
    -- Snapshot taken when the sheet opened. Frozen deliberately: recomputing it
    -- at post time would fold later movements into the variance and blame the counter.
    system_qty   numeric(18,4) not null default 0,
    counted_qty  numeric(18,4) not null default 0,
    variance_qty numeric(18,4) generated always as (counted_qty - system_qty) stored,
    unit_cost    numeric(20,6) not null default 0,
    remarks      varchar(250),
    row_version  bigint        not null default 1,
    constraint uq_inv_count_line unique (count_no, line_no),
    primary key (count_dtl_no)
);
```
